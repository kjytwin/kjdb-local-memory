cmake_minimum_required(VERSION 3.20)
project(KJDBLocalMemory VERSION 0.1.1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(KJDB_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(kjdb_memory SHARED "${KJDB_ROOT}/src/memory_store.c")
option(KJM_TESTING "Build the crash-injection library for acceptance tests" OFF)
if(KJM_TESTING)
  target_compile_definitions(kjdb_memory PRIVATE KJM_TESTING)
endif()
target_include_directories(kjdb_memory PRIVATE "${KJDB_ROOT}/include")
if(WIN32)
  if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This Windows package requires MSVC x64")
  endif()
  set(WIN_DIR "${KJDB_ROOT}/src/platform/windows")
  target_sources(kjdb_memory PRIVATE "${WIN_DIR}/windows_compat.c")
  target_include_directories(kjdb_memory PRIVATE "${WIN_DIR}")
  target_compile_options(kjdb_memory PRIVATE /W4 /WX /utf-8 "/FI${WIN_DIR}/windows_compat.h")
  target_compile_definitions(kjdb_memory PRIVATE _CRT_SECURE_NO_WARNINGS _CRT_RAND_S _WIN32_WINNT=0x0602)
  target_link_libraries(kjdb_memory PRIVATE advapi32)
  set_target_properties(kjdb_memory PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_definitions(kjdb_memory PRIVATE _POSIX_C_SOURCE=200809L _FILE_OFFSET_BITS=64)
  target_compile_options(kjdb_memory PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
